# XoPhim P3 (User-State: Watchlist + History / Continue-Watching) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the USER-STATE layer — Drizzle tables (`watchlist`, `watch_progress`), Fastify routes under `/v1/me/*` (all auth-guarded), and a typed web data-layer (TanStack Query hooks with guest-store branching) — so the web app can persist and read a user's watchlist and watch history/continue-watching.

**Architecture:** Two Postgres tables referenced by `movie_slug` (no FK to a catalog table; catalog is never persisted). A single Fastify plugin at `/v1/me` enforces `requireAuth` on every route. The web hooks branch on auth state: logged-in → server; logged-out → guest `localStorage` store inherited from Phase 2.

**Tech Stack:** Drizzle ORM + postgres-js, Fastify 5 with `fastify-type-provider-zod`, Zod 3, Vitest (unit + `app.inject` integration), React 18 + TanStack Query.

**Depends on Phase 2:** `app.requireAuth` preHandler, `request.user = { id, role, username, email }` (module-augmented onto `FastifyRequest`), `useAuth` hook, and the guest store at `apps/web/src/lib/guest-store.ts`. Do NOT re-implement auth; treat it as a precondition.

```ts
  index,
  integer,
  jsonb,
  pgTable,
  text,
  timestamp,
  unique,
  uuid,
} from "drizzle-orm/pg-core";

// ---------------------------------------------------------------------------
// Shared snapshot shape stored in jsonb columns.
// Allows list views (watchlist, history rails) to render without re-fetching
// KKPhim per item. Accept mild staleness; refresh on detail open.
// ---------------------------------------------------------------------------
export interface MovieSnapshot {
  name: string;
  posterUrl: string;
  type: string;
  year: number | null;
}

// ---------------------------------------------------------------------------
// watchlist
// ---------------------------------------------------------------------------
export const watchlist = pgTable(
  "watchlist",
  {
    id: uuid("id").primaryKey().defaultRandom(),
    user_id: uuid("user_id").notNull(),
    movie_slug: text("movie_slug").notNull(),
    movie_snapshot: jsonb("movie_snapshot").$type<MovieSnapshot>().notNull(),
    created_at: timestamp("created_at", { withTimezone: true }).notNull().defaultNow(),
  },
  (t) => ({
    uniq_user_movie: unique("watchlist_user_id_movie_slug_key").on(t.user_id, t.movie_slug),
    idx_user: index("watchlist_user_id_idx").on(t.user_id),
  }),
);

// ---------------------------------------------------------------------------
// watch_progress  (Xem Tiếp / History)
// ---------------------------------------------------------------------------
export const watchProgress = pgTable(
  "watch_progress",
  {
    id: uuid("id").primaryKey().defaultRandom(),
    user_id: uuid("user_id").notNull(),
    movie_slug: text("movie_slug").notNull(),
    // 'full' for phim lẻ (single movies), otherwise the KKPhim episode slug.
    episode_slug: text("episode_slug").notNull(),
    server_name: text("server_name").notNull().default(""),
    position_sec: integer("position_sec").notNull().default(0),
    duration_sec: integer("duration_sec"),
    movie_snapshot: jsonb("movie_snapshot").$type<MovieSnapshot>().notNull(),
    updated_at: timestamp("updated_at", { withTimezone: true }).notNull().defaultNow(),
  },
  (t) => ({
    uniq_user_movie_ep: unique("watch_progress_user_movie_ep_key").on(
      t.user_id,
      t.movie_slug,
      t.episode_slug,
    ),
    idx_user_updated: index("watch_progress_user_updated_idx").on(t.user_id, t.updated_at),
  }),
);
```

```ts
// shouldOverwriteProgress
// Returns true only when the incoming update is strictly newer than the stored

describe("shouldOverwriteProgress", () => {

    expect(shouldOverwriteProgress(t, t)).toBe(false);
  });

  it("returns false when incoming is older", () => {
    const stored = new Date("2026-01-01T12:00:00Z");
    const incoming = new Date("2026-01-01T10:00:00Z");
    expect(shouldOverwriteProgress(stored, incoming)).toBe(false);

  it("returns true when stored is null (first write)", () => {
    const incoming = new Date("2026-01-01T10:00:00Z");
    expect(shouldOverwriteProgress(null, incoming)).toBe(true);
  });

// first) — the Xem Tiếp rail shows the top items.
// ---------------------------------------------------------------------------
describe("sortByUpdatedAt", () => {

    const rows = [
      { id: "a", updated_at: new Date("2026-01-01T08:00:00Z") },
      { id: "b", updated_at: new Date("2026-01-01T12:00:00Z") },
      { id: "c", updated_at: new Date("2026-01-01T10:00:00Z") },
```

```ts
// Pure business-logic functions for the user-state routes.
// Extracted here so they can be unit-tested without Fastify or Postgres.

/**
 * Returns true when the incoming progress update should overwrite the stored
 * row. Only allow overwrite when `incoming` is strictly after `stored` — this
 * prevents a stale client (e.g. mobile after being offline) from rolling back a
 * newer desktop position. Consistent with P2 merge-guest semantics.
 */
export function shouldOverwriteProgress(
  stored: Date | null,
  incoming: Date,
): boolean {
  if (stored === null) return true;
  return incoming.getTime() > stored.getTime();
}

/**
 * Sort an array of objects that have `updated_at: Date` descending (newest
 * first). Returns a new array; does not mutate the input.
 */
export function sortByUpdatedAt<T extends { updated_at: Date }>(rows: T[]): T[] {
  return [...rows].sort((a, b) => b.updated_at.getTime() - a.updated_at.getTime());
}
```

## Task 3: Fastify user-state routes plugin

**Files:**
- Create: `apps/api/src/user-state/routes.ts`
- Modify: `apps/api/src/routes.ts`

All routes live under `/v1/me` and carry `{ preHandler: app.requireAuth }`. The plugin reads `request.user.id` (provided by Phase 2) and calls `app.db` (the decorated Drizzle instance).

Progress UPSERT uses Postgres `ON CONFLICT … DO UPDATE` guarded by the `shouldOverwriteProgress` check — only sets the new values when the incoming `updated_at` is strictly newer, preserving merge-guest semantics from Phase 2.

- [ ] **Step 1: Create the routes plugin**

Create `apps/api/src/user-state/routes.ts`:
```ts
import { eq, and, desc } from "drizzle-orm";
import type { FastifyPluginAsyncZod } from "fastify-type-provider-zod";
import { z } from "zod";

import { watchlist, watchProgress } from "../db/schema/index.js";
import { shouldOverwriteProgress } from "./logic.js";

// ---------------------------------------------------------------------------
// Shared Zod schemas
// ---------------------------------------------------------------------------

const movieSnapshotSchema = z.object({
  name: z.string(),
  posterUrl: z.string(),
  type: z.string(),
  year: z.number().nullable(),
});

const watchlistItemSchema = z.object({
  id: z.string().uuid(),
  movie_slug: z.string(),
  movie_snapshot: movieSnapshotSchema,
  created_at: z.string().datetime(),
});

const progressItemSchema = z.object({
  id: z.string().uuid(),
  movie_slug: z.string(),
  episode_slug: z.string(),
  server_name: z.string(),
  position_sec: z.number().int(),
  duration_sec: z.number().int().nullable(),
  movie_snapshot: movieSnapshotSchema,
  updated_at: z.string().datetime(),
});

// ---------------------------------------------------------------------------
// Plugin
// ---------------------------------------------------------------------------

export const registerUserStateRoutes: FastifyPluginAsyncZod = async (app) => {
  // -------------------------------------------------------------------------
  // GET /v1/me/watchlist
  // -------------------------------------------------------------------------
  app.get(
    "/watchlist",
    {
      preHandler: app.requireAuth,
      schema: {
        response: {
          200: z.object({ items: z.array(watchlistItemSchema) }),
        },
      },
    },
    async (request) => {
      const rows = await app.db
        .select()
        .from(watchlist)
        .where(eq(watchlist.user_id, request.user.id))
        .orderBy(desc(watchlist.created_at));

      return {
        items: rows.map((r) => ({
          id: r.id,
          movie_slug: r.movie_slug,
          movie_snapshot: r.movie_snapshot,
          created_at: r.created_at.toISOString(),
        })),
      };
    },
  );

  // -------------------------------------------------------------------------
  // PUT /v1/me/watchlist/:slug  — add to watchlist
  // -------------------------------------------------------------------------
  app.put(
    "/watchlist/:slug",
    {
      preHandler: app.requireAuth,
      schema: {
        params: z.object({ slug: z.string().min(1) }),
        body: z.object({ movie_snapshot: movieSnapshotSchema }),
        response: {
          200: watchlistItemSchema,
        },
      },
    },
    async (request, reply) => {
      const { slug } = request.params;
      const { movie_snapshot } = request.body;

      const [existing] = await app.db
        .select()
        .from(watchlist)
        .where(
          and(
            eq(watchlist.user_id, request.user.id),
            eq(watchlist.movie_slug, slug),
          ),
        )
        .limit(1);

      if (existing) {
        return reply.code(200).send({
          id: existing.id,
          movie_slug: existing.movie_slug,
          movie_snapshot: existing.movie_snapshot,
          created_at: existing.created_at.toISOString(),
        });
      }

      const [inserted] = await app.db
        .insert(watchlist)
        .values({
          user_id: request.user.id,
          movie_slug: slug,
          movie_snapshot,
        })
        .returning();

      if (!inserted) {
        const err = new Error("Failed to add to watchlist");
        (err as unknown as { statusCode: number }).statusCode = 500;
        throw err;
      }

      return reply.code(200).send({
        id: inserted.id,
        movie_slug: inserted.movie_slug,
        movie_snapshot: inserted.movie_snapshot,
        created_at: inserted.created_at.toISOString(),
      });
    },
  );

  // -------------------------------------------------------------------------
  // DELETE /v1/me/watchlist/:slug  — remove from watchlist
  // -------------------------------------------------------------------------
  app.delete(
    "/watchlist/:slug",
    {
      preHandler: app.requireAuth,
      schema: {
        params: z.object({ slug: z.string().min(1) }),
        response: {
          200: z.object({ ok: z.literal(true) }),
        },
      },
    },
    async (request, reply) => {
      await app.db
        .delete(watchlist)
        .where(
          and(
            eq(watchlist.user_id, request.user.id),
            eq(watchlist.movie_slug, request.params.slug),
          ),
        );
      return reply.code(200).send({ ok: true });
    },
  );

  // -------------------------------------------------------------------------
  // GET /v1/me/history  — full history ordered by updated_at desc
  //   Items with position_sec > 0 also serve as the "Xem Tiếp" (continue-watching) rail.
  // -------------------------------------------------------------------------
  app.get(
    "/history",
    {
      preHandler: app.requireAuth,
      schema: {
        response: {
          200: z.object({ items: z.array(progressItemSchema) }),
        },
      },
    },
    async (request) => {
      const rows = await app.db
        .select()
        .from(watchProgress)
        .where(eq(watchProgress.user_id, request.user.id))
        .orderBy(desc(watchProgress.updated_at));

      return {
        items: rows.map((r) => ({
          id: r.id,
          movie_slug: r.movie_slug,
          episode_slug: r.episode_slug,
          server_name: r.server_name,
          position_sec: r.position_sec,
          duration_sec: r.duration_sec ?? null,
          movie_snapshot: r.movie_snapshot,
          updated_at: r.updated_at.toISOString(),
        })),
      };
    },
  );

  // -------------------------------------------------------------------------
  // PUT /v1/me/progress  — upsert watch progress
  //   UPSERT on unique(user_id, movie_slug, episode_slug).
  //   Only overwrites when incoming updated_at is strictly newer (see logic.ts).
  // -------------------------------------------------------------------------
  app.put(
    "/progress",
    {
      preHandler: app.requireAuth,
      schema: {
        body: z.object({
          slug: z.string().min(1),
          episodeSlug: z.string().min(1),
          server: z.string().default(""),
          positionSec: z.number().int().min(0),
          durationSec: z.number().int().min(0).nullable().optional(),
          snapshot: movieSnapshotSchema,
        }),
        response: {
          200: progressItemSchema,
        },
      },
    },
    async (request, reply) => {
      const { slug, episodeSlug, server, positionSec, durationSec, snapshot } =
        request.body;
      const userId = request.user.id;
      const now = new Date();

      // Check existing row to enforce no-overwrite-newer.
      const [existing] = await app.db
        .select({ updated_at: watchProgress.updated_at })
        .from(watchProgress)
        .where(
          and(
            eq(watchProgress.user_id, userId),
            eq(watchProgress.movie_slug, slug),
            eq(watchProgress.episode_slug, episodeSlug),
          ),
        )
        .limit(1);

      if (existing && !shouldOverwriteProgress(existing.updated_at, now)) {
        // Return the existing row unchanged — incoming is not newer.
        const [current] = await app.db
          .select()
          .from(watchProgress)
          .where(
            and(
              eq(watchProgress.user_id, userId),
              eq(watchProgress.movie_slug, slug),
              eq(watchProgress.episode_slug, episodeSlug),
            ),
          )
          .limit(1);
        if (!current) {
          const err = new Error("Progress row disappeared");
          (err as unknown as { statusCode: number }).statusCode = 500;
          throw err;
        }
        return reply.code(200).send({
          id: current.id,
          movie_slug: current.movie_slug,
          episode_slug: current.episode_slug,
          server_name: current.server_name,
          position_sec: current.position_sec,
          duration_sec: current.duration_sec ?? null,
          movie_snapshot: current.movie_snapshot,
          updated_at: current.updated_at.toISOString(),
        });
      }

      // Upsert: insert or update when incoming is newer.
      const [upserted] = await app.db
        .insert(watchProgress)
        .values({
          user_id: userId,
          movie_slug: slug,
          episode_slug: episodeSlug,
          server_name: server,
          position_sec: positionSec,
          duration_sec: durationSec ?? null,
          movie_snapshot: snapshot,
          updated_at: now,
        })
        .onConflictDoUpdate({
          target: [
            watchProgress.user_id,
            watchProgress.movie_slug,
            watchProgress.episode_slug,
          ],
          set: {
            server_name: server,
            position_sec: positionSec,
            duration_sec: durationSec ?? null,
            movie_snapshot: snapshot,
            updated_at: now,
          },
        })
        .returning();

      if (!upserted) {
        const err = new Error("Failed to upsert progress");
        (err as unknown as { statusCode: number }).statusCode = 500;
        throw err;
      }

      return reply.code(200).send({
        id: upserted.id,
        movie_slug: upserted.movie_slug,
        episode_slug: upserted.episode_slug,
        server_name: upserted.server_name,
        position_sec: upserted.position_sec,
        duration_sec: upserted.duration_sec ?? null,
        movie_snapshot: upserted.movie_snapshot,
        updated_at: upserted.updated_at.toISOString(),
      });
    },
  );

  // -------------------------------------------------------------------------
  // DELETE /v1/me/history/:slug  — remove all progress for a movie
  // -------------------------------------------------------------------------
  app.delete(
    "/history/:slug",
    {
      preHandler: app.requireAuth,
      schema: {
        params: z.object({ slug: z.string().min(1) }),
        response: {
          200: z.object({ ok: z.literal(true) }),
        },
      },
    },
    async (request, reply) => {
      await app.db
        .delete(watchProgress)
        .where(
          and(
            eq(watchProgress.user_id, request.user.id),
            eq(watchProgress.movie_slug, request.params.slug),
          ),
        );
      return reply.code(200).send({ ok: true });
    },
  );
};
```

## Task 6: Web TanStack Query hooks with guest-store branching

**Files:**
- Create: `apps/web/src/hooks/user-state.ts`

All hooks branch on auth state:
- **Logged in:** read/write via the server (`userStateApi`).
- **Logged out (guest):** read/write the guest store in `localStorage` (`apps/web/src/lib/guest-store.ts`, provided by Phase 2).

`useToggleWatchlist` performs an optimistic update: it flips the local cache immediately, fires the server mutation, and rolls back on error. `useSaveProgress` is called by the player every ~15 seconds and on pause — the throttle is left to the caller (e.g. a `useInterval` / `useEffect` with a ref to the last-saved timestamp), but a helper `shouldThrottleProgressSave` is exported for the player component to use.

Phase 2 provides `useAuth()` which returns `{ user: AuthUser | null }`. The hooks use `user !== null` as the auth-state branch condition.

```ts
import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { useAuth } from "./auth"; // provided by Phase 2
import { userStateApi } from "../lib/user-state-api";
import { guestStore } from "../lib/guest-store"; // provided by Phase 2
import type { MovieSnapshot, SaveProgressPayload, WatchlistItem } from "../lib/user-state-types";

// ---------------------------------------------------------------------------
// Query keys
// ---------------------------------------------------------------------------
export const userStateKeys = {
  watchlist: ["user-state", "watchlist"] as const,
  history: ["user-state", "history"] as const,
};

// ---------------------------------------------------------------------------
// Throttle helper — exported so the player component can decide whether to
// call useSaveProgress. Saves are suppressed if the last save was < 15s ago.
// ---------------------------------------------------------------------------
const PROGRESS_THROTTLE_MS = 15_000;

export function shouldThrottleProgressSave(lastSavedAt: number | null): boolean {
  if (lastSavedAt === null) return false; // no prior save → allow
  return Date.now() - lastSavedAt < PROGRESS_THROTTLE_MS;
}

// ---------------------------------------------------------------------------
// useWatchlist
// Logged in → fetches from server.
// Guest      → reads from guestStore.watchlist (Zustand + localStorage).
// ---------------------------------------------------------------------------
export function useWatchlist() {
  const { user } = useAuth();

  // Server path.
  const serverQuery = useQuery({
    queryKey: userStateKeys.watchlist,
    queryFn: () => userStateApi.getWatchlist(),
    enabled: !!user,
    staleTime: 30_000,
    select: (data) => data.items,
  });

  // Guest path: read directly from the Zustand store.
  const guestItems = guestStore((s) => s.watchlist);

  if (user) {
    return {
      items: serverQuery.data ?? [],
      isLoading: serverQuery.isLoading,
      error: serverQuery.error,
    };
  }
  return { items: guestItems, isLoading: false, error: null };
}

// ---------------------------------------------------------------------------
// useToggleWatchlist
// Logged in → PUT/DELETE server + optimistic update.
// Guest      → toggle in guestStore (localStorage, no server call).
// ---------------------------------------------------------------------------
export function useToggleWatchlist() {
  const { user } = useAuth();
  const qc = useQueryClient();

  const serverMutation = useMutation({
    mutationFn: async ({
      slug,
      snapshot,
      currentlyInWatchlist,
    }: {
      slug: string;
      snapshot: MovieSnapshot;
      currentlyInWatchlist: boolean;
    }) => {
      if (currentlyInWatchlist) {
        await userStateApi.removeFromWatchlist(slug);
      } else {
        await userStateApi.addToWatchlist(slug, snapshot);
      }
    },
    // Optimistic update: flip the item in the cache before the server responds.
    onMutate: async ({ slug, snapshot, currentlyInWatchlist }) => {
      await qc.cancelQueries({ queryKey: userStateKeys.watchlist });
      const previous = qc.getQueryData<{ items: WatchlistItem[] }>(
        userStateKeys.watchlist,
      );
      qc.setQueryData<{ items: WatchlistItem[] }>(
        userStateKeys.watchlist,
        (old) => {
          const items = old?.items ?? [];
          if (currentlyInWatchlist) {
            return { items: items.filter((i) => i.movie_slug !== slug) };
          }
          const optimistic: WatchlistItem = {
            id: `optimistic-${slug}`,
            movie_slug: slug,
            movie_snapshot: snapshot,
            created_at: new Date().toISOString(),
          };
          return { items: [optimistic, ...items] };
        },
      );
      return { previous };
    },
    // On error: roll back to the previous cache value.
    onError: (_err, _vars, context) => {
      if (context?.previous !== undefined) {
        qc.setQueryData(userStateKeys.watchlist, context.previous);
      }
    },
    // Always refetch after settle to sync with server truth.
    onSettled: () => {
      qc.invalidateQueries({ queryKey: userStateKeys.watchlist });
    },
  });

  // Guest path.
  const toggleGuest = guestStore((s) => s.toggleWatchlist);

  return {
    toggle: (slug: string, snapshot: MovieSnapshot, currentlyInWatchlist: boolean) => {
      if (user) {
        serverMutation.mutate({ slug, snapshot, currentlyInWatchlist });
      } else {
        toggleGuest(slug, snapshot);
      }
    },
    isPending: serverMutation.isPending,
    error: serverMutation.error,
  };
}

// ---------------------------------------------------------------------------
// useHistory
// Logged in → fetches from server (ordered desc by updated_at, per API).
// Guest      → reads from guestStore.progress (localStorage).
// ---------------------------------------------------------------------------
export function useHistory() {
  const { user } = useAuth();

  const serverQuery = useQuery({
    queryKey: userStateKeys.history,
    queryFn: () => userStateApi.getHistory(),
    enabled: !!user,
    staleTime: 30_000,
    select: (data) => data.items,
  });

  const guestProgress = guestStore((s) => s.progress);

  if (user) {
    return {
      items: serverQuery.data ?? [],
      isLoading: serverQuery.isLoading,
      error: serverQuery.error,
    };
  }
  // Guest progress is already ordered by the store (Zustand keeps insertion order).
  return { items: guestProgress, isLoading: false, error: null };
}

// ---------------------------------------------------------------------------
// useSaveProgress
// Logged in → PUT /v1/me/progress.
// Guest      → upsert in guestStore (localStorage, merge-guest semantics).
//
// The player component is responsible for throttling calls (use
// shouldThrottleProgressSave to gate saves at the call site — ~15s or on pause).
// ---------------------------------------------------------------------------
export function useSaveProgress() {
  const { user } = useAuth();
  const qc = useQueryClient();
  const saveGuest = guestStore((s) => s.saveProgress);

  const serverMutation = useMutation({
    mutationFn: (payload: SaveProgressPayload) => userStateApi.saveProgress(payload),
    onSuccess: () => {
      // Invalidate history so the Xem Tiếp rail refreshes.
      qc.invalidateQueries({ queryKey: userStateKeys.history });
    },
  });

  return {
    save: (payload: SaveProgressPayload) => {
      if (user) {
        serverMutation.mutate(payload);
      } else {
        saveGuest(payload);
      }
    },
    isPending: serverMutation.isPending,
    error: serverMutation.error,
  };
}
```
